package main

import (
	"fmt"
	"math/rand"
)

func ri(inf, sup int) float64 {
	return float64(rand.Intn(sup-inf+1) + inf)
}

func embua(pen *Pen, dist float64) {
	if dist < 5 {
		return
	}

	pen.SetLineWidth(dist / 45)
	pen.Walk(dist)
	pen.Right(90)
	dist *= 0.97

	embua(pen, dist)
}

func main() {
	pen := NewPen(800, 800)
	pen.SetHeading(0)
	pen.SetPosition(200, 200)
	embua(pen, 400)
	pen.SavePNG("embua.png")
	fmt.Println("PNG file created successfully.")
}
