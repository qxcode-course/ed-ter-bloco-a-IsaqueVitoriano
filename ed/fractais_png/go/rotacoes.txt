package main

import (
	"fmt"
	"math/rand"
)

func ri(inf, sup int) float64 {
	return float64(rand.Intn(sup-inf+1) + inf)
}

func rotacoes(pen *Pen, dist float64) {
	if dist < 15 {
		return
	}

	pen.SetRGB(ri(0, 255), ri(0, 255), ri(0, 255))
	pen.SetLineWidth(dist / 45.0)
	pen.Walk(dist)
	pen.Right(90)
	dist *= 0.969

	rotacoes(pen, dist)
}

func main() {
	pen := NewPen(200, 200)
	pen.SetHeading(0)
	pen.SetPosition(25, 25)
	rotacoes(pen, 150)
	pen.SavePNG("rotacoes.png")
	fmt.Println("PNG file created successfully.")
}
